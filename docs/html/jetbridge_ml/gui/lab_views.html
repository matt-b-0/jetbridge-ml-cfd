 12def export_views(output, payload, held, studies, metrics):
 13    """Save scientific figures and a self-contained interactive HTML dashboard.
 14
 15    Parameters
 16    ----------
 17    output : pathlib.Path
 18        Existing isolated destination.
 19    payload : dict
 20        Coordinates, full-array predictions and evidence metadata.
 21    held, studies, metrics : pandas.DataFrame
 22        Held-location predictions, sensitivity trials and common-hull metrics.
 23    """
 24    plt.rcParams.update({'font.family': 'DejaVu Sans', 'font.size': 10,
 25                         'axes.spines.top': False, 'axes.spines.right': False,
 26                         'axes.titleweight': 'bold', 'axes.labelcolor': '#334155',
 27                         'text.color': '#0f172a', 'axes.edgecolor': '#cbd5e1',
 28                         'figure.facecolor': '#f8fafc', 'axes.facecolor': '#ffffff'})
 29    def save(fig, name):
 30        for ext in ('png', 'pdf'):
 31            fig.savefig(output/f'{name}.{ext}', dpi=180, bbox_inches='tight')
 32        plt.close(fig)
 33    q=np.asarray(payload['query']); sensors=np.asarray(payload['sensors'])
 34    nx,ny=payload['nx'],payload['ny']; floor=payload['floor']
 35    data=[np.asarray(payload['fields'][m],float) for m in METHODS]
 36    lo=min(np.nanmin(v) for v in data); hi=max(np.nanmax(v) for v in data)
 37    norm=Normalize(lo,hi); hull=np.asarray(payload['inside'])
 38    fig,axes=plt.subplots(1,4,figsize=(13,6),layout='constrained',sharex=True,sharey=True)
 39    for ax,method,values in zip(axes,METHODS,data):
 40        image=ax.pcolormesh(q[:,0].reshape(ny,nx),(q[:,1]-floor).reshape(ny,nx),
 41                            values.reshape(ny,nx),cmap='viridis',norm=norm,shading='nearest')
 42        ax.contourf(q[:,0].reshape(ny,nx),(q[:,1]-floor).reshape(ny,nx),
 43                    (~hull).reshape(ny,nx),levels=[.5,1.5],colors='none',hatches=['////'])
 44        ax.scatter(sensors[:,0],sensors[:,1]-floor,s=20,c='white',edgecolors='#0f172a',lw=.7)
 45        ax.set(title=LABELS[method],xlabel='x (m)',xlim=(0,payload['width']),ylim=(0,payload['height']))
 46        ax.set_aspect('equal')
 47    axes[0].set_ylabel('Height above local floor (m)')
 48    fig.colorbar(image,ax=axes,label='Scalar speed (m/s)',shrink=.7)
 49    fig.suptitle(payload['title']+'\nSame sensors and colour scale | full-array refits; hatching = outside sensor hull')
 50    save(fig,'01_method_comparison')
 51    if 'reference' in payload:
 52        reference=np.asarray(payload['reference'])
 53        fig,axes=plt.subplots(1,5,figsize=(15,5.5),layout='constrained',sharey=True)
 54        ref=axes[0].pcolormesh(q[:,0].reshape(ny,nx),(q[:,1]-floor).reshape(ny,nx),
 55                               reference.reshape(ny,nx),cmap='viridis',shading='nearest')
 56        axes[0].set_title('Archived CFD reference')
 57        fig.colorbar(ref,ax=axes[0],label='Speed (m/s)',shrink=.65)
 58        bound=max(np.nanmax(np.abs(v-reference)) for v in data)
 59        for ax,m,v in zip(axes[1:],METHODS,data):
 60            image=ax.pcolormesh(q[:,0].reshape(ny,nx),(q[:,1]-floor).reshape(ny,nx),
 61                                (v-reference).reshape(ny,nx),cmap='RdBu_r',vmin=-bound,vmax=bound,shading='nearest')
 62            ax.set_title(LABELS[m]+' − CFD')
 63        fig.colorbar(image,ax=list(axes[1:]),label='Signed difference (m/s)',shrink=.65)
 64        for ax in axes:ax.set_aspect('equal');ax.set_xlabel('x (m)')
 65        axes[0].set_ylabel('Height above floor (m)')
 66        fig.suptitle('Where does sparse reconstruction lose the numerical field?\nFull display includes extrapolation; common-support scores use disjoint test points only')
 67        save(fig,'01b_virtual_field_errors')
 68    common=held[held.inside_hull & held.method.isin(METHODS)]
 69    fig,axes=plt.subplots(1,4,figsize=(13,4),layout='constrained')
 70    limit=[min(common.speed_mps.min(),common.prediction_mps.min()),
 71           max(common.speed_mps.max(),common.prediction_mps.max())]
 72    margin=max(.005,(limit[1]-limit[0])*.1);limit=[limit[0]-margin,limit[1]+margin]
 73    for ax,m in zip(axes,METHODS):
 74        part=common[common.method==m]
 75        ax.plot(limit,limit,'--',color='#94a3b8',lw=1)
 76        ax.scatter(part.speed_mps,part.prediction_mps,c=COLORS[m],s=35,alpha=.8)
 77        metric=metrics[metrics.method==m].iloc[0]
 78        ax.set(xlim=limit,ylim=limit,xlabel='Held-out reference (m/s)',
 79               title=f'{LABELS[m]}\nRMSE {metric.rmse:.4f} m/s')
 80        ax.set_aspect('equal');ax.grid(alpha=.15)
 81    axes[0].set_ylabel('Prediction (m/s)')
 82    fig.suptitle('Predict locations the model did not see\nCommon supported locations; separate from the full-array maps')
 83    save(fig,'02_heldout_predictions')
 84    fig,ax=plt.subplots(figsize=(11,4.5),layout='constrained')
 85    for i,m in enumerate(METHODS):
 86        part=held[held.method==m].sort_values('source_row')
 87        ax.scatter(part.source_row+(i-1.5)*.12,part.prediction_mps-part.speed_mps,
 88                   c=COLORS[m],label=LABELS[m],s=32,
 89                   marker=['o','s','^','D'][i])
 90    ax.axhline(0,c='#94a3b8',lw=1);ax.legend(ncol=4,loc='upper center',bbox_to_anchor=(.5,1.2))
 91    ax.set(xlabel='Source sensor row / virtual test index (zero based)',ylabel='Prediction − reference (m/s)',
 92           title='Local error reveals where the reconstruction misses')
 93    save(fig,'02b_local_errors')
 94    clean=studies[studies.noise_fraction==0]
 95    fig,ax=plt.subplots(figsize=(10,5),layout='constrained')
 96    for m in METHODS:
 97        g=clean[clean.method==m].groupby('sensor_count').rmse
 98        mean=g.mean();low=g.min();high=g.max()
 99        ax.plot(mean.index,mean.values,'o-',color=COLORS[m],label=LABELS[m],lw=2)
100        ax.fill_between(mean.index,low.values,high.values,color=COLORS[m],alpha=.12)
101    ax.set(xlabel='Retained training sensors',ylabel='Common-support held-out RMSE (m/s)',
102           title='How much information do more sensors add?\nThree nested layouts; fixed hull anchors and fixed held-out targets')
103    ax.legend(ncol=2);ax.grid(alpha=.15);save(fig,'03_sensor_learning')
104    noisy=studies[studies.sensor_count==studies.sensor_count.max()]
105    fig,ax=plt.subplots(figsize=(10,5),layout='constrained')
106    for m in METHODS:
107        g=noisy[noisy.method==m].groupby('noise_fraction').rmse
108        a=g.mean();ax.plot(a.index*100,a.values,'o-',color=COLORS[m],label=LABELS[m],lw=2)
109        ax.fill_between(a.index*100,g.min().values,g.max().values,color=COLORS[m],alpha=.12)
110    ax.set(xlabel='Gaussian noise σ / retained-speed RMS (%)',ylabel='Common-support held-out RMSE (m/s)',
111           title='How robust is the fit to noisy inputs?\nTraining targets perturbed; held-out references unchanged; bands = trial range')
112    ax.legend(ncol=2);ax.grid(alpha=.15);save(fig,'04_noise_sensitivity')
113    fig,ax=plt.subplots(figsize=(10,5),layout='constrained')
114    timing=payload['timing']
115    for m in METHODS:
116        point=metrics[metrics.method==m].iloc[0]
117        seconds=np.median([row['seconds'] for row in timing if row['method']==m])
118        ax.scatter(seconds*1000,point.rmse,c=COLORS[m],s=130,edgecolors='white',lw=1.5)
119        ax.annotate(LABELS[m],(seconds*1000,point.rmse),xytext=(9,6),textcoords='offset points',color=COLORS[m])
120    ax.set(xscale='log',xlabel=f'Median full fit + {len(q):,} grid predictions (ms; log scale)',
121           ylabel='Common-support held-location RMSE (m/s)',
122           title='Is the extra fitting cost worthwhile?\nSVR includes training-only search; accuracy comes from separate held-out fits')
123    ax.margins(x=.3,y=.25);ax.grid(alpha=.15);save(fig,'05_accuracy_cost')
124    payload['metrics']=metrics.replace({np.nan:None}).to_dict('records')
125    payload['studies']=studies.to_dict('records')
126    (output/'dashboard_data.json').write_text(json.dumps(payload,allow_nan=False))
127    template=HTML.replace('__TITLE__',html.escape(payload['title']))
128    # Escape script delimiters even though IDs and values are controlled inputs.
129    encoded=json.dumps(payload,allow_nan=False).replace('<','\\u003c')
130    (output/'reconstruction_lab.html').write_text(template.replace('__PAYLOAD__',encoded),encoding='utf-8')
