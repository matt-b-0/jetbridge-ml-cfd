 32def verify_inputs(root, spec=None, cancel_file=None):
 33    """Hash-check exactly the bundled sources read by this job.
 34
 35    Parameters
 36    ----------
 37    root : pathlib.Path
 38        Checkout containing data/demo and its hash manifest.
 39    spec : RunSpec, optional
 40        Request whose declared inputs are checked; None checks the full bundle.
 41    cancel_file : pathlib.Path, optional
 42        Job-specific cancellation marker, checked between files.
 43
 44    Returns
 45    -------
 46    dict
 47        Verified input digests and the bundle manifest digest.
 48    """
 49    bundle = root / 'data/demo'
 50    manifest_path = bundle / 'manifest.json'
 51    records = json.loads(manifest_path.read_text())['files']
 52    if spec is None:
 53        selected = records
 54    else:
 55        needed = {'experimental/Data Dictionary.csv',
 56                  f'experimental/03-Averages/V-{spec.run_id}.txt'}
 57        prefixes = ()
 58        if spec.kind == 'fresh':
 59            needed.update({'experimental/06-Periods/Periods.csv',
 60                           f'experimental/05-HVAC/{spec.run_id}.csv'})
 61        if spec.kind == 'lab_cfd':
 62            needed = {'virtual_samples/dense_test.csv', 'virtual_samples/sampling_manifest.json',
 63                      'cases_metadata/iso/log.simpleFoam'}
 64        if spec.kind == 'archive':
 65            needed.update({'reference/matched/matched_sensor_audit.csv','cases_metadata/iso/log.simpleFoam'})
 66        if spec.kind == 'thermal':
 67            prefixes = ('cases_metadata/thermal/', 'cases_metadata/control/', 'thermal_comparisons/')
 68        selected = [r for r in records if r['path'] in needed or r['path'].startswith(prefixes)]
 69        if not needed.issubset({r['path'] for r in selected}):
 70            raise ValueError('Requested inputs are absent from the bundled hash manifest')
 71    verified = {}
 72    for record in selected:
 73        if cancel_file is not None and cancel_file.exists():
 74            raise Cancelled('Cancelled while verifying selected inputs')
 75        path = (bundle / record['path']).resolve()
 76        if not path.is_relative_to(bundle.resolve()) or not path.is_file():
 77            raise ValueError('Invalid or missing bundled input')
 78        digest = sha256(path.read_bytes()).hexdigest()
 79        if digest != record['sha256']:
 80            raise ValueError(f'Changed bundled input: {record["path"]}')
 81        verified[record['path']] = digest
 82    return dict(files=verified, bundle_manifest_sha256=sha256(manifest_path.read_bytes()).hexdigest())

 85def boundary_inputs(root, run_id):
 86    """Read the selected run's entire-set HVAC window in SI units.
 87
 88    Parameters
 89    ----------
 90    root : pathlib.Path
 91        Repository with the attributed experimental subset.
 92    run_id : str
 93        Supported single-window S01-A-HN experiment.
 94
 95    Returns
 96    -------
 97    dict
 98        Flow, temporal coverage and temperature context. No test targets fit Q.
 99    """
100    import numpy as np
101    import pandas as pd
102    dataset = root / 'data/demo/experimental'
103    periods = pd.read_csv(dataset / '06-Periods/Periods.csv', encoding='cp1252')
104    match = periods[periods.Set == run_id]
105    if len(match) != 1:
106        raise ValueError('Fresh CFD requires one unambiguous measured HVAC window')
107    row = match.iloc[0]
108    path = dataset / f'05-HVAC/{run_id}.csv'
109    table = pd.read_csv(path, encoding='cp1252').replace(-9999, np.nan)
110    start, end = float(row['Entire Set-Start (sec)']), float(row['Entire Set-End (sec)'])
111    window = table[table['Time (sec)'].between(start, end)]
112    q = window['Q.FS (lps)']; q = q[np.isfinite(q) & (q >= 0)]
113    if not len(q) or q.mean() <= 0:
114        raise ValueError('No positive measured supply flow in the source window')
115    result = dict(flow_m3s=float(q.mean() / 1000), valid_hvac_readings=len(q),
116                  entire_start_sec=start, entire_end_sec=end,
117                  averaging='Arithmetic mean of nonnegative valid Q.FS in Entire Set window',
118                  source_sha256=sha256(path.read_bytes()).hexdigest(),
119                  windows_equal=bool(start == row['Stable-Start (sec)'] and end == row['Stable-End (sec)']))
120    for source, key in [('T.db.SA (°C)', 'supply_C'), ('Tdb.Lab (°C)', 'lab_C')]:
121        values = window[source].dropna()
122        result[key] = float(values.mean()) if len(values) else None
123    return result

126def prepare_case(root, output, frame, mesh, q):
127    """Prepare a cold-start isothermal case without modifying the template.
128
129    Parameters
130    ----------
131    root, output : pathlib.Path
132        Repository and new case destination.
133    frame : pandas.DataFrame
134        One experiment's absolute metre sensor coordinates.
135    mesh : str
136        Named resolution from MESHES; all use uniform grading.
137    q : float
138        Positive measured supply flow in cubic metres per second.
139
140    Returns
141    -------
142    dict
143        Mesh counts, initial velocity scale and declared turbulence assumptions.
144    """
145    import numpy as np
146    from jetbridge_ml.cfd_interface.plot_comparison import mesh_profile
147    source = root / 'cfd/cases/empty_bridge_s01_a_10_hn_1'
148    output.mkdir(parents=True, exist_ok=False)
149    for folder in ('0', 'constant', 'system'):
150        shutil.copytree(source / folder, output / folder, ignore=shutil.ignore_patterns('polyMesh'))
151    path = output / 'system/blockMeshDict'
152    nx, ny, *nz = MESHES[mesh]; index = iter(nz)
153    text, count = re.subn(r'(hex\s*\([^)]*\)\s*)\(\d+\s+\d+\s+\d+\)',
154                         lambda m: m[1] + f'({nx} {ny} {next(index)})', path.read_text())
155    if count != 3:
156        raise ValueError('Expected exactly three baseline mesh blocks')
157    path.write_text(text)
158    width, zs, floors, tops = mesh_profile(path)
159    xyz = frame[['x_m', 'y_m', 'z_m']].to_numpy()
160    local = xyz[:, 1] - np.interp(xyz[:, 2], zs, floors)
161    if (np.any(xyz[:, 0] <= 0) or np.any(xyz[:, 0] >= width) or np.any(local <= 0)
162            or np.any(local >= np.interp(xyz[:, 2], zs, tops - floors))
163            or np.any(xyz[:, 2] <= zs[0]) or np.any(xyz[:, 2] >= zs[-1])):
164        raise ValueError('Experimental sensor falls outside the CFD domain')
165    height = tops[0] - floors[0]
166    bulk = q / (width * height)
167    dh = 2 * width * height / (width + height)
168    k = 1.5 * (.05 * bulk) ** 2
169    epsilon = .09 ** .75 * k ** 1.5 / (.07 * dh)
170    u = output / '0/U'; text = u.read_text()
171    text = re.sub(r'(volumetricFlowRate\s+constant\s+)[\d.eE+-]+',
172                  lambda m: m[1] + f'{q:.12g}', text)
173    text = re.sub(r'uniform\s*\(\s*0\s+0\s+-[\d.eE+]+\s*\)',
174                  f'uniform (0 0 {-bulk:.12g})', text)
175    u.write_text(text)
176    for field, value in [('k', k), ('epsilon', epsilon)]:
177        p = output / '0' / field
178        p.write_text(re.sub(r'(uniform\s+)[\d.eE+-]+(\s*;)',
179                            lambda m: m[1] + f'{value:.12g}' + m[2], p.read_text()))
180    path = output / 'system/controlDict'; text = path.read_text()
181    text = re.sub(r'endTime\s+\d+\s*;', 'endTime 600;', text)
182    points = '\n'.join('(' + ' '.join(f'{v:.12g}' for v in row) + ')' for row in xyz)
183    text, count = re.subn(r'probeLocations\s*\(.*?\);', 'probeLocations (\n' + points + '\n);', text, flags=re.S)
184    if count != 1:
185        raise ValueError('Expected one baseline probe array')
186    path.write_text(text)
187    return dict(cells=cell_count(mesh), counts=MESHES[mesh], bulk_speed_mps=bulk,
188                initial_k_m2s2=k, initial_epsilon_m2s3=epsilon,
189                intensity_assumption=.05, length_scale_m=.07 * dh,
190                physics='steady isothermal k-epsilon RANS; full-face inlet; sealed no-slip walls',
191                mesh_resolution_is_not_accuracy=True, iteration_cap=600)

230    def command(self, name, command, cwd=None):
231        """Stream a subprocess into a log and stop its Linux group on cancellation."""
232        self.check_cancel(); print(f'STAGE: {name}', flush=True)
233        record = dict(name=name, command=list(map(str, command)), status='running')
234        self.manifest['stages'].append(record); self.save(); start = perf_counter()
235        env = dict(os.environ, PYTHONPATH=str(self.root / 'src'), PYTHONUNBUFFERED='1')
236        tracker = None
237        with (self.output / f'log.{name}').open('w') as handle:
238            proc = subprocess.Popen(command, cwd=cwd or self.root, env=env, stdout=subprocess.PIPE,
239                                    stderr=subprocess.STDOUT, text=True, bufsize=1, start_new_session=True)
240            if name == 'simpleFoam':
241                self.manifest.update(fresh_cfd=True, solver_attempted=True)
242                self.save()
243            try:
244                if name == 'simpleFoam':
245                    tracker = ResidualRecorder(self.output / 'residual_history.csv')
246                stream = queue.Queue()
247                def read_output():
248                    for line in proc.stdout:
249                        stream.put(line)
250                    stream.put(None)
251                threading.Thread(target=read_output, daemon=True).start()
252                while True:
253                    self.check_cancel()
254                    try:
255                        line = stream.get(timeout=.2)
256                    except queue.Empty:
257                        continue
258                    if line is None:
259                        break
260                    handle.write(line); handle.flush()
261                    if tracker is not None:tracker.feed(line)
262                    if line.startswith('Time = '):
263                        if int(float(line.split('=')[1])) % 25 == 0:
264                            print(line.strip() + ' (SIMPLE iteration)', flush=True)
265                    elif (('converged' in line.lower() and name == 'simpleFoam') or 'Mesh OK' in line or 'ERROR' in line):
266                        print(line.strip(), flush=True)
267                if proc.wait() != 0:
268                    raise RuntimeError(f'{name} failed. See log.{name}')
269            except Cancelled:
270                record['status'] = 'cancelled'
271                raise
272            except Exception:
273                record['status'] = 'failed'
274                raise
275            finally:
276                if proc.poll() is None:
277                    os.killpg(proc.pid, signal.SIGTERM)
278                    try:
279                        proc.wait(timeout=10)
280                    except subprocess.TimeoutExpired:
281                        os.killpg(proc.pid, signal.SIGKILL); proc.wait()
282                if tracker is not None:
283                    tracker.close()
284                    self.manifest['residual_samples'] = tracker.count
285                record['elapsed_seconds'] = round(perf_counter() - start, 3)
286        record['status'] = 'complete'; self.save()

293    def sample(self, case, saved, points):
294        """Export matched CFD points and final inlet/outlet flux, with alignment checks."""
295        import numpy as np
296        from scipy.spatial import cKDTree
297        positions = '\n'.join('(' + ' '.join(f'{v:.12g}' for v in row) + ')' for row in points)
298        text = ('FoamFile { version 2.0; format ascii; class dictionary; object guiSample; } functions { '
299                'guiPoints { type sets; libs ("libsampling.so"); writeControl writeTime; setFormat raw; '
300                'interpolationScheme cellPoint; fields (U); sets (points { type cloud; axis xyz; points ('
301                + positions + '); }); } ')
302        for name, patch in [('guiIn', 'inlet'), ('guiOut', 'outlet')]:
303            text += (f'{name} {{ type surfaceFieldValue; libs ("libfieldFunctionObjects.so"); '
304                     f'regionType patch; name {patch}; operation sum; writeFields false; fields (phi); }} ')
305        dictionary = self.output / 'sample.dict'; dictionary.write_text(text + '}')
306        self.foam('sampling', ['simpleFoam', '-case', case, '-postProcess', '-time', saved, '-dict', dictionary], case)
307        files = list((case / f'postProcessing/guiPoints/{saved}').glob('points_U.*'))
308        if len(files) != 1:
309            raise ValueError('Cannot identify sampled velocity file')
310        raw = np.loadtxt(files[0]); distance, index = cKDTree(raw[:, :3]).query(points)
311        if raw.shape != (len(points), 6) or not np.isfinite(raw).all() or max(distance) > 1e-6 or len(set(index)) != len(points):
312            raise ValueError('CFD coordinate/field alignment failed')
313        log = (self.output / 'log.sampling').read_text()
314        qin = float(re.findall(r'sum\(inlet\) of phi = ([\d.eE+-]+)', log)[-1])
315        qout = float(re.findall(r'sum\(outlet\) of phi = ([\d.eE+-]+)', log)[-1])
316        imbalance = abs(qin + qout) / abs(qin)
317        self.manifest['flux'] = dict(inlet_m3s=qin, outlet_m3s=qout, relative_imbalance=imbalance,
318                                      declared_screen=1e-5)
319        if imbalance > 1e-5:
320            raise RuntimeError('Final flux imbalance exceeds the declared numerical screen')
321        return np.linalg.norm(raw[index, 3:], axis=1)

345def execute(root, output, spec):
346    """Run one request in a fresh folder with explicit data lineage.
347
348    Parameters
349    ----------
350    root, output : pathlib.Path
351        Repository and new result destination.
352    spec : RunSpec
353        Supported experiment, run type, reconstruction and mesh selection.
354
355    Returns
356    -------
357    dict
358        Completed run manifest. Failure and cancellation also retain manifests.
359    """
360    import numpy as np
361    import pandas as pd
362    from jetbridge_ml.data_processing.faa import read_speed_average
363    from jetbridge_ml.cfd_interface.plot_comparison import mesh_profile
364    from .plots import reconstruction, mesh_plot, cfd_comparison
365    spec.validate(root); output = validate_output(root, output, fresh_cfd=spec.kind == 'fresh')
366    if spec.kind == 'fresh' and (os.name == 'nt' or not FOAM.is_file()):
367        raise ValueError('Fresh CFD requires Linux/WSL and OpenCFD v2412')
368    if spec.kind == 'fresh':
369        validate_resources(spec.processes)
370    output.mkdir(parents=True)
371    job = Job(root, output, spec)
372    originals = root / 'cfd/cases/empty_bridge_s01_a_10_hn_1'
373    hashes = {str(p.relative_to(root)): sha256(p.read_bytes()).hexdigest()
374              for folder in ('0', 'constant', 'system') for p in (originals / folder).rglob('*')
375              if p.is_file() and 'polyMesh' not in p.parts}
376    job.manifest['original_case_sha256'] = hashes
377    try:
378        print('Checking selected bundled input hashes...', flush=True)
379        verified = verify_inputs(root, spec, job.cancel_file)
380        job.manifest['verified_input_files'] = len(verified['files'])
381        job.manifest['verified_inputs'] = verified
382        job.check_cancel()
383        if spec.kind == 'archive':
384            tracker = ResidualRecorder(output / 'residual_history.csv','archived isothermal CFD')
385            try:
386                with (root / 'data/demo/cases_metadata/iso/log.simpleFoam').open() as source:
387                    for line in source:tracker.feed(line)
388            finally:tracker.close()
389            job.manifest['residual_samples'] = tracker.count
390        elif spec.kind == 'thermal':
391            import csv
392            records = read_history(root / 'data/demo/thermal_comparisons/thermal_residual_history.csv')
393            with (output / 'residual_history.csv').open('w',newline='') as handle:
394                writer=csv.DictWriter(handle,fieldnames=COLUMNS);writer.writeheader()
395                for row in records:
396                    row['source']='archived unconverged thermal CFD';writer.writerow(row)
397            job.manifest['residual_samples']=len(records)
398        dataset = root / 'data/demo/experimental'
399        mesh = originals / 'system/blockMeshDict'
400        if spec.kind in ('lab', 'lab_cfd'):
401            from .lab import build_lab
402            if spec.kind == 'lab_cfd':
403                source_log = (root / 'data/demo/cases_metadata/iso/log.simpleFoam').read_text()
404                if not re.search(r'SIMPLE solution converged in 237 iterations', source_log):
405                    raise ValueError('Archived CFD convergence evidence is missing')
406            job.manifest['reconstruction_lab'] = build_lab(root, output, spec, job.check_cancel)
407        elif spec.kind == 'thermal':
408            meta = root / 'data/demo/cases_metadata'
409            job.manifest.update(numerical_status='archived NOT CONVERGED', interpolation='not applied',
410                                training_reference_eligible=False)
411            job.command('thermal_diagnostics', [sys.executable, '-m', 'jetbridge_ml.cfd_interface.plot_thermal',
412                        '--case', str(meta / 'thermal'), '--control', str(meta / 'control'),
413                        '--replay-inputs', str(root / 'data/demo/thermal_comparisons'),
414                        '--output', str(output / 'thermal')])
415        else:
416            frame = read_speed_average(dataset / f'03-Averages/V-{spec.run_id}.txt')
417            job.manifest['experimental_file_sha256'] = sha256((dataset / f'03-Averages/V-{spec.run_id}.txt').read_bytes()).hexdigest()
418            job.manifest['sensor_count'] = len(frame)
419            width, zs, floors, tops = mesh_profile(mesh)
420            z = float(frame.z_m.iloc[0]); floor = float(np.interp(z, zs, floors))
421            height = float(np.interp(z, zs, tops) - floor)
422            xx, hh = np.meshgrid((np.arange(40)+.5)*width/40, (np.arange(60)+.5)*height/60)
423            query = np.column_stack([xx.ravel(), hh.ravel() + floor])
424            print('Reconstructing experimental plane and evaluating withheld sensors...', flush=True)
425            fit = reconstruction(frame, query, spec.method, floor, output)
426            job.manifest['experimental_reconstruction'] = fit
427            job.check_cancel()
428            if spec.kind == 'fresh':
429                boundary = boundary_inputs(root, spec.run_id)
430                job.manifest['boundary'] = boundary
431                case = output / 'case'
432                job.manifest['case_setup'] = prepare_case(root, case, frame, spec.mesh, boundary['flow_m3s'])
433                mesh_plot(case / 'system/blockMeshDict', frame, spec.mesh, output)
434                job.save()
435                job.foam('blockMesh', ['blockMesh'], case)
436                job.foam('checkMesh', ['checkMesh'], case)
437                mesh_ok = 'Mesh OK' in (output / 'log.checkMesh').read_text()
438                job.manifest['mesh_ok'] = mesh_ok
439                if not mesh_ok:
440                    raise RuntimeError('checkMesh failed; CFD comparison rejected')
441                job.manifest['execution'] = dict(processes=spec.processes, available_cpus=available_cpus(),
442                                                 decomposition='scotch' if spec.processes > 1 else None)
443                if spec.processes > 1:
444                    job.foam('mpi_version', ['mpirun', '--version'], case)
445                    write_decomposition(case, spec.processes)
446                    job.foam('decomposePar', ['decomposePar'], case)
447                    job.foam('parallelCheckMesh', mpi_command(spec.processes, 'checkMesh'), case)
448                    if 'Mesh OK' not in (output / 'log.parallelCheckMesh').read_text():
449                        raise RuntimeError('Parallel checkMesh failed; comparison rejected')
450                command = mpi_command(spec.processes, 'simpleFoam') if spec.processes > 1 else ['simpleFoam']
451                job.foam('simpleFoam', command, case)
452                match = re.search(r'SIMPLE solution converged in (\d+) iterations', (output / 'log.simpleFoam').read_text())
453                if not match:
454                    job.manifest['solver_converged'] = False
455                    raise RuntimeError('Iteration cap reached without convergence; CFD comparison rejected')
456                saved = match[1]
